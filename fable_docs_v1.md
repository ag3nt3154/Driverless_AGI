def _write_session(path: Path, records: list[dict]) -> None:
    path.write_text(
        "\n".join(json.dumps(r) for r in records) + "\n",
        encoding="utf-8",
        newline="\n",
    )


def _minimal_session(total_in=1000, total_out=200, cost=0.01, tools=None) -> list[dict]:
    return [
        {"type": "session_start", "model": "gpt-4o", "started_at": "2026-07-06T00:00:00Z"},
        {"type": "message", "entity": "system", "content": "x" * 400},
        {"type": "message", "entity": "user", "content": "do the thing"},
        {
            "type": "message", "entity": "assistant",
            "input_tokens": total_in, "output_tokens": total_out, "cost": cost,
            "tool_calls": [],
        },
        {
            "type": "session_end",
            "total_input_tokens": total_in, "total_output_tokens": total_out,
            "total_cost": cost, "tool_call_counts": tools or {},
        },
    ]

#!/usr/bin/env python
"""scripts/token_report.py — Token usage analysis for dagi session logs.

Usage:
    conda run -n dagi python scripts/token_report.py
    conda run -n dagi python scripts/token_report.py --logs .dagi/logs --last 10
    conda run -n dagi python scripts/token_report.py --json
"""
from __future__ import annotations

import argparse
import json
import statistics
from pathlib import Path


def parse_session(path: Path) -> dict | None:
    """Parse one session JSONL file. Returns None on any parse error."""
    try:
        text = path.read_text(encoding="utf-8")
    except OSError:
        return None
    lines = [l for l in text.splitlines() if l.strip()]
    if not lines:
        return None
    try:
        records = [json.loads(l) for l in lines]
    except json.JSONDecodeError:
        return None

    stats: dict = {
        "path": str(path),
        "model": None,
        "started_at": None,
        "iterations": 0,
        "compactions": 0,
        "total_input_tokens": 0,
        "total_output_tokens": 0,
        "total_cost": None,
        "tool_counts": {},
        "estimated_system_tokens": 0,
        "input_tokens_per_turn": [],
    }

    for rec in records:
        t = rec.get("type")
        if t == "session_start":
            stats["model"] = rec.get("model")
            stats["started_at"] = rec.get("started_at")
        elif t == "session_end":
            stats["total_input_tokens"] = rec.get("total_input_tokens") or 0
            stats["total_output_tokens"] = rec.get("total_output_tokens") or 0
            stats["total_cost"] = rec.get("total_cost")
            stats["tool_counts"] = rec.get("tool_call_counts") or {}
        elif t == "message":
            entity = rec.get("entity", "")
            if entity == "system":
                content = rec.get("content") or ""
                stats["estimated_system_tokens"] = len(content) // 4
            elif entity == "assistant":
                stats["iterations"] += 1
                in_tok = rec.get("input_tokens")
                if in_tok:
                    stats["input_tokens_per_turn"].append(int(in_tok))
            elif entity == "user":
                content = rec.get("content") or ""
                if content.startswith("[CONTEXT SUMMARY"):
                    stats["compactions"] += 1
    return stats


def aggregate_sessions(sessions: list[dict]) -> dict:
    """Compute aggregate stats across a list of parse_session results."""
    valid = [s for s in sessions if s is not None]
    if not valid:
        return {"count": 0}

    input_tokens = [s["total_input_tokens"] for s in valid]
    output_tokens = [s["total_output_tokens"] for s in valid]
    iterations = [s["iterations"] for s in valid]
    costs = [s["total_cost"] for s in valid if s["total_cost"] is not None]
    compactions = [s["compactions"] for s in valid]

    # Aggregate tool counts
    all_tools: dict[str, int] = {}
    for s in valid:
        for name, count in (s["tool_counts"] or {}).items():
            all_tools[name] = all_tools.get(name, 0) + count

    return {
        "count": len(valid),
        "avg_input_tokens": statistics.mean(input_tokens) if input_tokens else 0,
        "median_input_tokens": statistics.median(input_tokens) if input_tokens else 0,
        "avg_output_tokens": statistics.mean(output_tokens) if output_tokens else 0,
        "avg_iterations": statistics.mean(iterations) if iterations else 0,
        "avg_compactions": statistics.mean(compactions) if compactions else 0,
        "total_cost": sum(costs) if costs else None,
        "tool_counts": dict(sorted(all_tools.items(), key=lambda x: -x[1])),
    }


def _print_report(agg: dict, sessions: list[dict], json_out: bool) -> None:
    if json_out:
        print(json.dumps({"aggregate": agg, "sessions": sessions}, indent=2, default=str))
        return

    print(f"\n=== dagi Token Report ({agg['count']} sessions) ===\n")
    print(f"  Avg input tokens/session : {agg['avg_input_tokens']:,.0f}")
    print(f"  Median input tokens      : {agg['median_input_tokens']:,.0f}")
    print(f"  Avg output tokens/session: {agg['avg_output_tokens']:,.0f}")
    print(f"  Avg iterations/session   : {agg['avg_iterations']:.1f}")
    print(f"  Avg compactions/session  : {agg['avg_compactions']:.2f}")
    if agg["total_cost"] is not None:
        print(f"  Total cost               : ${agg['total_cost']:.4f}")
    print()

    if agg["tool_counts"]:
        print("  Tool call totals (top 10):")
        for name, count in list(agg["tool_counts"].items())[:10]:
            print(f"    {name:<30} {count:>6}")
    print()

    print("  Per-session breakdown:")
    header = f"  {'Model':<20} {'In':>8} {'Out':>7} {'Iter':>5} {'Compact':>7} {'Cost':>8}"
    print(header)
    print("  " + "-" * (len(header) - 2))
    for s in sessions:
        if s is None:
            continue
        cost_str = f"${s['total_cost']:.4f}" if s["total_cost"] is not None else "  n/a "
        model = (s["model"] or "unknown")[:20]
        print(
            f"  {model:<20} "
            f"{s['total_input_tokens']:>8,} "
            f"{s['total_output_tokens']:>7,} "
            f"{s['iterations']:>5} "
            f"{s['compactions']:>7} "
            f"{cost_str:>8}"
        )
    print()


def main() -> None:
    parser = argparse.ArgumentParser(description="dagi token usage report")
    parser.add_argument("--logs", default=".dagi/logs", help="Path to session logs directory")
    parser.add_argument("--last", type=int, default=0, help="Only analyze the N most recent sessions (0 = all)")
    parser.add_argument("--json", action="store_true", help="Output raw JSON instead of table")
    args = parser.parse_args()

    logs_dir = Path(args.logs)
    if not logs_dir.exists():
        print(f"Logs directory not found: {logs_dir}")
        return

    paths = sorted(logs_dir.glob("session_*.jsonl"), key=lambda p: p.stat().st_mtime, reverse=True)
    if args.last > 0:
        paths = paths[: args.last]

    sessions = [parse_session(p) for p in paths]
    agg = aggregate_sessions([s for s in sessions if s is not None])
    _print_report(agg, sessions, args.json)


if __name__ == "__main__":
    main()
```

"""benchmarks/smoke/assert_utils.py — Programmatic assertion helpers for smoke tasks."""
from __future__ import annotations
import subprocess
from pathlib import Path


class AssertionError(Exception):
    """Raised when a smoke task assertion fails."""


def check_assertion(assertion: dict, project_dir: Path) -> None:
    """Run a single assertion dict against project_dir. Raises AssertionError on failure.

    Supported types:
        file_exists       path (str) — file must exist under project_dir
        file_contains     path (str), text (str) — file must contain text
        file_not_contains path (str), text (str) — file must NOT contain text
        pytest_passes     (no extra fields) — `pytest` in project_dir must exit 0
    """
    t = assertion.get("type")
    if t == "file_exists":
        target = project_dir / assertion["path"]
        if not target.exists():
            raise AssertionError(f"file_exists FAILED: {assertion['path']} not found in {project_dir}")
    elif t == "file_contains":
        target = project_dir / assertion["path"]
        if not target.exists():
            raise AssertionError(f"file_contains FAILED: {assertion['path']} not found")
        content = target.read_text(encoding="utf-8")
        if assertion["text"] not in content:
            raise AssertionError(
                f"file_contains FAILED: '{assertion['text']}' not found in {assertion['path']}"
            )
    elif t == "file_not_contains":
        target = project_dir / assertion["path"]
        if not target.exists():
            raise AssertionError(f"file_not_contains FAILED: {assertion['path']} not found")
        content = target.read_text(encoding="utf-8")
        if assertion["text"] in content:
            raise AssertionError(
                f"file_not_contains FAILED: '{assertion['text']}' found in {assertion['path']}"
            )
    elif t == "pytest_passes":
        result = subprocess.run(
            ["conda", "run", "-n", "dagi", "pytest", "--tb=short", "-q"],
            cwd=project_dir,
            capture_output=True,
            text=True,
            timeout=120,
        )
        if result.returncode != 0:
            raise AssertionError(
                f"pytest_passes FAILED (exit {result.returncode}):\n{result.stdout[-2000:]}"
            )
    else:
        raise ValueError(f"Unknown assertion type: {t!r}")
```

"""benchmarks/smoke/runner.py — L1 smoke eval harness for dagi.

Runs each YAML task against a real model and records results to a JSONL file.
Always run from the dagi repo root; session logs are written to .dagi/logs/.

Usage:
    conda run -n dagi python benchmarks/smoke/runner.py
    conda run -n dagi python benchmarks/smoke/runner.py --tasks benchmarks/smoke/tasks
    conda run -n dagi python benchmarks/smoke/runner.py --model gpt-4o-mini-openai --runs 3
"""
from __future__ import annotations

import argparse
import datetime
import json
import os
import subprocess
import sys
import tempfile
import time
from pathlib import Path

import yaml

sys.path.insert(0, str(Path(__file__).parent.parent.parent / "scripts"))
from token_report import parse_session

_DAGI_ROOT = Path(__file__).parent.parent.parent


def load_task(path: Path) -> dict:
    """Parse a smoke task YAML file."""
    return yaml.safe_load(path.read_text(encoding="utf-8"))


def write_fixtures(fixture_files: dict[str, str], project_dir: Path) -> None:
    """Write fixture file contents into project_dir, creating subdirs as needed."""
    for rel_path, content in fixture_files.items():
        target = project_dir / rel_path
        target.parent.mkdir(parents=True, exist_ok=True)
        target.write_text(content, encoding="utf-8", newline="\n")


def parse_latest_session_log(logs_dir: Path, after: datetime.datetime) -> dict | None:
    """Find the most recent session log written after `after` and parse it."""
    candidates = [
        p for p in logs_dir.glob("session_*.jsonl")
        if datetime.datetime.fromtimestamp(p.stat().st_mtime, tz=datetime.timezone.utc) > after
    ]
    if not candidates:
        return None
    latest = max(candidates, key=lambda p: p.stat().st_mtime)

    # Inline minimal parsing to avoid circular import of full token_report
    try:
        lines = [l for l in latest.read_text(encoding="utf-8").splitlines() if l.strip()]
        records = [json.loads(l) for l in lines]
    except (OSError, json.JSONDecodeError):
        return None

    stats: dict = {
        "log_path": str(latest),
        "total_input_tokens": 0,
        "total_output_tokens": 0,
        "total_cost": None,
        "iterations": 0,
        "tool_counts": {},
    }
    for rec in records:
        t = rec.get("type")
        if t == "session_end":
            stats["total_input_tokens"] = rec.get("total_input_tokens") or 0
            stats["total_output_tokens"] = rec.get("total_output_tokens") or 0
            stats["total_cost"] = rec.get("total_cost")
            stats["tool_counts"] = rec.get("tool_call_counts") or {}
        elif t == "message" and rec.get("entity") == "assistant":
            stats["iterations"] += 1
    return stats


def record_result(result: dict, output_path: Path) -> None:
    """Append a result dict as a JSONL line."""
    with open(output_path, "a", encoding="utf-8", newline="\n") as fh:
        fh.write(json.dumps(result, default=str) + "\n")


def run_task(
    task: dict,
    model_id: str,
    max_iter: int,
    logs_dir: Path,
    results_path: Path,
    run_index: int = 0,
) -> dict:
    """Run one smoke task. Returns the result dict."""
    with tempfile.TemporaryDirectory(prefix="smoke_") as tmp_dir:
        project_dir = Path(tmp_dir)
        write_fixtures(task.get("fixture_files") or {}, project_dir)

        start_time = datetime.datetime.now(tz=datetime.timezone.utc)
        wall_start = time.monotonic()

        proc = subprocess.run(
            [
                "conda", "run", "-n", "dagi",
                "python", str(_DAGI_ROOT / "main.py"),
                task["prompt"],
                "--model", model_id,
                "--max-iter", str(max_iter),
                "--project", str(project_dir),
            ],
            cwd=_DAGI_ROOT,
            capture_output=True,
            text=True,
            timeout=300,
        )
        wall_time = time.monotonic() - wall_start

        session_stats = parse_latest_session_log(logs_dir, after=start_time)

        # Run assertions
        success = proc.returncode == 0
        assertion_errors: list[str] = []
        if success:
            from assert_utils import check_assertion, AssertionError as SmokeAssertionError
            for assertion in (task.get("assertions") or []):
                try:
                    check_assertion(assertion, project_dir)
                except SmokeAssertionError as exc:
                    assertion_errors.append(str(exc))
                    success = False

        result = {
            "task": task["name"],
            "run_index": run_index,
            "model": model_id,
            "success": success,
            "wall_time": round(wall_time, 2),
            "assertion_errors": assertion_errors,
            "exit_code": proc.returncode,
            "total_input_tokens": (session_stats or {}).get("total_input_tokens", 0),
            "total_output_tokens": (session_stats or {}).get("total_output_tokens", 0),
            "total_cost": (session_stats or {}).get("total_cost"),
            "iterations": (session_stats or {}).get("iterations", 0),
            "timestamp": start_time.isoformat(),
        }
        record_result(result, results_path)
        status = "PASS" if success else "FAIL"
        print(f"  [{status}] {task['name']} run {run_index} — {result['total_input_tokens']:,} tokens, {wall_time:.0f}s")
        return result


def main() -> None:
    parser = argparse.ArgumentParser(description="dagi L1 smoke eval harness")
    parser.add_argument("--tasks", default="benchmarks/smoke/tasks", help="Directory of task YAML files")
    parser.add_argument("--model", default=None, help="Model ID from config.yaml (default: config default_model)")
    parser.add_argument("--runs", type=int, default=1, help="Runs per task (use 3 for variance-aware comparison)")
    parser.add_argument("--max-iter", type=int, default=30, help="Max agent iterations per task")
    parser.add_argument("--output", default=None, help="Path for results JSONL (default: auto-timestamped)")
    args = parser.parse_args()

    tasks_dir = Path(args.tasks)
    task_files = sorted(tasks_dir.glob("*.yaml"))
    if not task_files:
        print(f"No task YAML files found in {tasks_dir}")
        sys.exit(1)

    logs_dir = _DAGI_ROOT / ".dagi" / "logs"
    ts = datetime.datetime.now().strftime("%Y-%m-%d_%H-%M-%S")
    results_dir = _DAGI_ROOT / "benchmarks" / "smoke" / "results"
    results_dir.mkdir(parents=True, exist_ok=True)
    results_path = Path(args.output) if args.output else results_dir / f"results_{ts}.jsonl"

    # Resolve model — if not given, read default_model from config.yaml
    model_id = args.model
    if not model_id:
        try:
            import yaml as _yaml
            cfg = _yaml.safe_load((_DAGI_ROOT / "config.yaml").read_text(encoding="utf-8"))
            model_id = cfg.get("worker_model") or cfg.get("default_model") or "gpt-4o-openai"
        except Exception:
            model_id = "gpt-4o-openai"

    print(f"\ndagi smoke harness — model: {model_id}, runs/task: {args.runs}")
    print(f"Tasks: {len(task_files)}  |  Output: {results_path}\n")

    for task_file in task_files:

                record_result({"task": task["name"], "run_index": i, "success": False, "error": str(exc)}, results_path)

    print(f"\nDone. Results: {results_path}")


if __name__ == "__main__":
    main()

"""benchmarks/smoke/ab_compare.py — A/B comparison for smoke run results.

Usage:
    conda run -n dagi python benchmarks/smoke/ab_compare.py baseline.jsonl candidate.jsonl
"""
from __future__ import annotations

import argparse
import json
from collections import defaultdict
from pathlib import Path

REGRESSION = "REGRESSION"
IMPROVEMENT = "IMPROVEMENT"
NO_CHANGE = "no_change"


def load_results(path: Path) -> dict[str, list[dict]]:
    """Load a results.jsonl file, grouped by task name."""
    groups: dict[str, list[dict]] = defaultdict(list)
    for line in path.read_text(encoding="utf-8").splitlines():
        line = line.strip()
        if not line:
            continue
        rec = json.loads(line)
        groups[rec["task"]].append(rec)
    return dict(groups)


def _pass_rate(runs: list[dict]) -> float:
    if not runs:
        return 0.0
    return sum(1 for r in runs if r.get("success")) / len(runs)


def _avg_tokens(runs: list[dict]) -> float:
    vals = [r.get("total_input_tokens", 0) for r in runs if r.get("total_input_tokens")]
    return sum(vals) / len(vals) if vals else 0.0


def compare(baseline: dict[str, list[dict]], candidate: dict[str, list[dict]]) -> dict[str, dict]:
    """Return per-task comparison dict keyed by task name."""
    all_tasks = set(baseline) | set(candidate)
    results: dict[str, dict] = {}

    for task in sorted(all_tasks):
        a_runs = baseline.get(task, [])
        b_runs = candidate.get(task, [])

        a_pass = _pass_rate(a_runs)
        b_pass = _pass_rate(b_runs)
        a_tokens = _avg_tokens(a_runs)
        b_tokens = _avg_tokens(b_runs)
        token_delta_pct = ((b_tokens - a_tokens) / a_tokens * 100) if a_tokens else 0.0

        if b_pass < a_pass:
            verdict = REGRESSION
        elif b_pass > a_pass:
            verdict = IMPROVEMENT
        else:
            verdict = NO_CHANGE

        results[task] = {
            "verdict": verdict,
            "baseline_pass_rate": a_pass,
            "candidate_pass_rate": b_pass,
            "baseline_avg_tokens": round(a_tokens),
            "candidate_avg_tokens": round(b_tokens),
            "token_delta_pct": round(token_delta_pct, 1),
        }
    return results


def main() -> None:
    parser = argparse.ArgumentParser(description="Compare two smoke eval result files")
    parser.add_argument("baseline", help="Baseline results.jsonl path")
    parser.add_argument("candidate", help="Candidate results.jsonl path")
    args = parser.parse_args()

    baseline = load_results(Path(args.baseline))
    candidate = load_results(Path(args.candidate))
    results = compare(baseline, candidate)

    regressions = [t for t, r in results.items() if r["verdict"] == REGRESSION]
    improvements = [t for t, r in results.items() if r["verdict"] == IMPROVEMENT]

    print("\n=== A/B Smoke Comparison ===\n")
    print(f"  {'Task':<30} {'Verdict':<12} {'Base%':>6} {'Cand%':>6} {'ΔTokens%':>10}")
    print("  " + "-" * 70)
    for task, r in results.items():
        verdict_str = r["verdict"]
        print(
            f"  {task:<30} {verdict_str:<12} "
            f"{r['baseline_pass_rate']*100:>5.0f}% "
            f"{r['candidate_pass_rate']*100:>5.0f}% "
            f"{r['token_delta_pct']:>+9.1f}%"
        )
    print()
    if regressions:
        print(f"  REGRESSIONS: {', '.join(regressions)}")
    if improvements:
        print(f"  IMPROVEMENTS: {', '.join(improvements)}")
    if not regressions and not improvements:
        print("  No changes in pass rate.")
    print()


if __name__ == "__main__":
    main()

"""tools/eviction.py — Stale tool-result eviction for AgentLoop.

Replaces tool-role messages older than `eviction_turns` assistant turns with
pointer stubs. The full content is written to a temp file so the agent can
re-read it on demand. This is a deterministic, zero-cost alternative to early
compaction for clearing dead-weight tool results from context.
"""
from __future__ import annotations

import os
import tempfile
from pathlib import Path

_ALREADY_FILTERED = "OUTPUT TRUNCATED"
_ALREADY_EVICTED = "[EVICTED"


def evict_stale_tool_results(
    messages: list[dict],
    eviction_turns: int,
    temp_dir: Path,
) -> int:
    """Replace tool-role messages older than eviction_turns assistant turns with stubs.

    Walks messages backwards, counting trailing assistant-role messages.
    Any tool-role message that has >= eviction_turns assistant messages after it
    is replaced in-place with a stub pointing to a temp file.

    Parameters
    ----------
    messages      : The live _messages list from AgentLoop (mutated in place).
    eviction_turns: Threshold. 0 = eviction disabled.
    temp_dir      : Directory for full-output temp files.

    Returns
    -------
    Number of messages evicted.
    """
    if eviction_turns <= 0:
        return 0

    temp_dir = Path(temp_dir)
    temp_dir.mkdir(parents=True, exist_ok=True)

    trailing_assistant = 0
    evicted = 0

    for i in range(len(messages) - 1, -1, -1):
        msg = messages[i]
        role = msg.get("role", "")

        if role == "assistant":
            trailing_assistant += 1
            continue

        if role != "tool":
            continue

        if trailing_assistant < eviction_turns:
            continue

        content = str(msg.get("content") or "")
        if _ALREADY_FILTERED in content or content.startswith(_ALREADY_EVICTED):
            continue

        try:
            fd, tmp_path = tempfile.mkstemp(dir=temp_dir, prefix="evicted_", suffix=".txt")
            os.close(fd)
            Path(tmp_path).write_text(content, encoding="utf-8", newline="\n")
            messages[i] = dict(msg, content=f"[EVICTED — stale tool result, full output at: {tmp_path}]")
            evicted += 1
        except OSError:
            pass  # fail open — leave message unchanged

    return evicted
```

#!/usr/bin/env python
"""scripts/mine_failures.py — Extract failure taxonomy from dagi session logs.

Failure tags are derived heuristically at read time (no runtime changes needed).

Usage:
    conda run -n dagi python scripts/mine_failures.py
    conda run -n dagi python scripts/mine_failures.py --logs .dagi/logs --last 50
    conda run -n dagi python scripts/mine_failures.py --json
"""
from __future__ import annotations

import argparse
import json
from collections import Counter
from enum import Enum
from pathlib import Path


class FailureTag(str, Enum):
    EDIT_MISMATCH = "edit_mismatch"       # edit tool "not found" errors
    LOOP_STALL    = "loop_stall"          # 3+ continuation-prompt injections
    TOOL_ERROR    = "tool_error"          # any tool_end with "Error" in result
    COMPACTION    = "compaction"          # at least one compaction occurred


_CONTINUE_MARKERS = ("please continue", "continue with the task", "continue.")
_LOOP_STALL_THRESHOLD = 3   # continuation injections before tagging as stall


def classify_session(path: Path) -> dict:
    """Read one session JSONL and return {path, tags, counts}."""
    try:
        lines = [l for l in path.read_text(encoding="utf-8").splitlines() if l.strip()]
        records = [json.loads(l) for l in lines]
    except (OSError, json.JSONDecodeError):
        return {"path": str(path), "tags": [], "counts": {}}

    tags: set[FailureTag] = set()
    continuation_count = 0
    seen_first_user = False

    for rec in records:
        t = rec.get("type")
        if t == "tool_end":
            name = rec.get("name", "")
            result = str(rec.get("result") or "").lower()
            if "error" in result:
                tags.add(FailureTag.TOOL_ERROR)
            if name == "edit" and "not found" in result:
                tags.add(FailureTag.EDIT_MISMATCH)
        elif t == "message":
            entity = rec.get("entity", "")
            content = str(rec.get("content") or "").strip().lower()
            if entity == "user":
                if content.startswith("[context summary"):
                    tags.add(FailureTag.COMPACTION)
                elif not seen_first_user:
                    seen_first_user = True
                elif any(m in content for m in _CONTINUE_MARKERS):
                    continuation_count += 1

    if continuation_count >= _LOOP_STALL_THRESHOLD:
        tags.add(FailureTag.LOOP_STALL)

    return {"path": str(path), "tags": sorted(t.value for t in tags), "counts": {"continuations": continuation_count}}


def main() -> None:
    parser = argparse.ArgumentParser(description="Mine failure taxonomy from session logs")
    parser.add_argument("--logs", default=".dagi/logs")
    parser.add_argument("--last", type=int, default=0)
    parser.add_argument("--json", action="store_true")
    args = parser.parse_args()

    logs_dir = Path(args.logs)
    paths = sorted(logs_dir.glob("session_*.jsonl"), key=lambda p: p.stat().st_mtime, reverse=True)
    if args.last > 0:
        paths = paths[: args.last]

    sessions = [classify_session(p) for p in paths]
    tag_counts = Counter(tag for s in sessions for tag in s["tags"])

    if args.json:
        print(json.dumps({"tag_counts": dict(tag_counts), "sessions": sessions}, indent=2))
        return

    print(f"\n=== dagi Failure Taxonomy ({len(sessions)} sessions) ===\n")
    for tag in FailureTag:
        count = tag_counts.get(tag.value, 0)
        pct = (count / len(sessions) * 100) if sessions else 0
        print(f"  {tag.value:<25} {count:>4} sessions  ({pct:.0f}%)")
    print()
    flagged = [s for s in sessions if s["tags"]]
    print(f"  Sessions with at least one tag: {len(flagged)}/{len(sessions)}")
    print()


if __name__ == "__main__":
    main()
```
